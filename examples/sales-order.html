完整代码 ​

ts
import { createPrintStudio } from '@printstack/print-studio'
import type { DesignStudio, PrintDocument } from '@printstack/print-studio'

export const orderInput = {
  order: { number: 'SO-2026-001', customer: '示例客户' },
  items: Array.from({ length: 45 }, (_, index) => {
    return { name: `商品 ${index + 1}`, quantity: 2, amount: (index + 1) * 10 }
  }),
}

export function buildOrder(studio: DesignStudio) {
  studio.document.create() // 从空白文档开始搭建（默认打开的欢迎模板仅用于设计器试用）
  studio.document.rename('订单与商品标签')
  const page = studio.page.active
  page.update({
    name: '订单',
    width: 210,
    height: 297,
    margins: { top: 25, right: 10, bottom: 20, left: 10 },
  })
  studio.dataSource.upsert({
    key: 'order',
    label: '订单',
    sampleData: orderInput.order,
    schema: [
      { name: 'number', type: 'string' },
      { name: 'customer', type: 'string' },
    ],
  })
  studio.dataSource.upsert({
    key: 'items',
    label: '商品明细',
    sampleData: orderInput.items.slice(0, 2),
    schema: [
      {
        name: 'item',
        type: 'array',
        children: [
          { name: 'name', type: 'string' },
          { name: 'quantity', type: 'number' },
          { name: 'amount', type: 'number' },
        ],
      },
    ],
  })
  const stationery = studio.template.create({ name: '公司抬头与页码' })
  stationery.page.update({
    width: 210,
    height: 297,
    margins: { top: 25, right: 10, bottom: 20, left: 10 },
  })
  stationery.page.elements.add({
    type: 'text',
    x: 10,
    y: 8,
    width: 190,
    height: 10,
    props: { text: '示例公司 · {{ $document.documentName }}', fontSize: 16 },
  })
  stationery.page.elements.add({
    type: 'text',
    x: 10,
    y: 283,
    width: 190,
    height: 8,
    props: {
      text: '第 {{ $document.pageNumber }} / {{ $document.pageCount }} 页',
      textAlign: 'center',
      fontSize: 9,
    },
  })
  studio.template.apply(stationery, [page])
  page.elements.add({
    type: 'text',
    x: 10,
    y: 27,
    width: 190,
    height: 10,
    props: { text: '订单：{{ order.number }}    客户：{{ order.customer }}' },
  })
  const table = page.elements.add({
    type: 'table',
    x: 10,
    y: 42,
    width: 190,
    bindings: { rows: 'items' },
    props: {
      columns: [
        { title: '商品', bind: 'name', width: 3, align: 'left' },
        { title: '数量', bind: 'quantity', width: 1, align: 'right' },
        {
          title: '金额',
          bind: 'amount',
          width: 2,
          align: 'right',
          format: { type: 'number', digits: 2 },
        },
      ],
      rows: [],
      heightMode: 'auto',
      pagination: {
        crossPage: true,
        repeatHeader: true,
        rowsPerPage: 20,
        fillEmptyRows: false,
        positionMode: 'flow',
      },
      summary: {
        enabled: true,
        position: 'bottom',
        scope: 'lastPage',
        columns: { '0': { type: 'text', text: '合计' }, '2': { type: 'sum' } },
      },
    },
  })
  const signature = page.elements.add({
    type: 'text',
    x: 10,
    y: 70,
    width: 120,
    height: 12,
    props: { text: '收货人签字：____________' },
  })
  const follow = studio.followBlock.create({
    pageId: page.id,
    elementIds: [signature.id],
    target: { kind: 'element', id: table.id },
    gap: 5,
    name: '签收区域',
  })
  if (!follow) {
    throw new Error('签收区域创建失败')
  }
  const labelPage = studio.page.create({
    name: '商品标签',
    width: 210,
    height: 297,
    margins: { top: 25, right: 10, bottom: 20, left: 10 },
  })
  studio.template.apply(stationery, [labelPage])
  const label = labelPage.elements.add({
    type: 'text',
    x: 10,
    y: 30,
    width: 55,
    height: 20,
    props: {
      text: '商品：{{ name }}\n数量：{{ quantity | number:0 }}',
      fontSize: 10,
    },
  })
  const labels = studio.repeatGroup.create({
    pageId: labelPage.id,
    elementIds: [label.id],
    name: '商品标签网格',
    collection: 'items',
    direction: 'grid',
    columns: 3,
    gapX: 5,
    gapY: 3,
    positionMode: 'flow',
  })
  if (!labels) {
    throw new Error('标签循环区块创建失败')
  }
  studio.page.setActive(page)
  return {
    page,
    stationery,
    table,
    signature,
    follow,
    labelPage,
    label,
    labels,
  }
}

export function createOrderDocument(): PrintDocument {
  const studio = createPrintStudio({ storage: false })
  try {
    buildOrder(studio)
    return studio.document.data
  } finally {
    studio.dispose()
  }
}

export async function renderOrder() {
  const studio = createPrintStudio({
    initialDocument: createOrderDocument(),
    storage: false,
  })
  try {
    const html = await studio.toHTML(orderInput)
    const pdf = await studio.toPDF(orderInput)
    return { document: studio.document.data, html, pdf }
  } finally {
    studio.dispose()
  }
}

关键 API 讲解 ​

API用途
template.create / apply公司抬头与页码跨页共用，渲染在内容层之下
bindings: { rows: 'items' }表格结构化绑定数组
pagination.crossPage + repeatHeader45 行自动分页且延续页带表头
summary.scope: 'lastPage'末页全量合计（'eachPage' 为每页小计）
followBlock.create签字区始终跟随表格末尾，gap: 5 mm
repeatGroup.create({ direction: 'grid' })标签页 3 列平铺

要点 ​

  • renderOrder() 演示「构建 → 取文档快照 → 重开实例装载 → 输出」的完整生命周期，对应业务上「设计态」与「运行态」分离。
  • initialDocument 装载已有文档时实例不重复构建模板。
  • 合计行对金额列求和（列索引 '2'），首列改为固定文本「合计」——columns 键为列索引字符串。
